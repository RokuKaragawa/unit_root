\section*{参数与标准素分解}
$p = 17$，$h = (p-1)/2 = 8$，模 $17$ 的最小原根 $g = 3$。

$h$ 的标准素分解：$h = 2\times2\times2$，子群链共 $3$ 层。

$m_0 = 1$，$m_{1} = 2$，$m_{2} = 4$，$m_{3} = 8$。

各层子群 $H_j = \{\pm g^{m_j t}\}$，其中 $H_0$ 是全部 $16$ 个非零剩余：

\fitmath{H_{1} = \{1,\,2,\,4,\,8,\,9,\,13,\,15,\,16\}}

\fitmath{H_{2} = \{1,\,4,\,13,\,16\}}

\fitmath{H_{3} = \{1,\,16\}}

各层周期 $\eta^{(j)}_{a} = \sum_{b \in H_j}\zeta^{ab}$（$a$ 取陪集代表）。

\section*{各层周期与分解表}
\subsection*{第 $0$ 层：1 个周期}
$H_0$ 是全部 $16$ 个非零剩余，故 $\eta^{(0)}_{1} = -1$。

\subsection*{第 $1$ 层：2 个周期}
\fitmath{\eta^{(1)}_{1} = \zeta^{1}+\zeta^{2}+\zeta^{4}+\zeta^{8}+\zeta^{9}+\zeta^{13}+\zeta^{15}+\zeta^{16}}

\fitmath{\eta^{(1)}_{3} = \zeta^{3}+\zeta^{5}+\zeta^{6}+\zeta^{7}+\zeta^{10}+\zeta^{11}+\zeta^{12}+\zeta^{14}}

\subsection*{第 $2$ 层：4 个周期}
\fitmath{\eta^{(2)}_{1} = \zeta^{1}+\zeta^{4}+\zeta^{13}+\zeta^{16}}

\fitmath{\eta^{(2)}_{2} = \zeta^{2}+\zeta^{8}+\zeta^{9}+\zeta^{15}}

\fitmath{\eta^{(2)}_{3} = \zeta^{3}+\zeta^{5}+\zeta^{12}+\zeta^{14}}

\fitmath{\eta^{(2)}_{6} = \zeta^{6}+\zeta^{7}+\zeta^{10}+\zeta^{11}}

\subsection*{第 $3$ 层：8 个周期}
\fitmath{\eta^{(3)}_{1} = \zeta^{1}+\zeta^{16}}

\fitmath{\eta^{(3)}_{2} = \zeta^{2}+\zeta^{15}}

\fitmath{\eta^{(3)}_{3} = \zeta^{3}+\zeta^{14}}

\fitmath{\eta^{(3)}_{4} = \zeta^{4}+\zeta^{13}}

\fitmath{\eta^{(3)}_{5} = \zeta^{5}+\zeta^{12}}

\fitmath{\eta^{(3)}_{6} = \zeta^{6}+\zeta^{11}}

\fitmath{\eta^{(3)}_{7} = \zeta^{7}+\zeta^{10}}

\subsection*{分解表：每个母周期拆成 $\ell_j$ 个子周期}
\fitmath{\eta^{(0)}_{1} = \eta^{(1)}_{1}+\eta^{(1)}_{3}}

（第 $1$ 层，$\ell = 2$。）

\fitmath{\eta^{(1)}_{1} = \eta^{(2)}_{1}+\eta^{(2)}_{2}}

\fitmath{\eta^{(1)}_{3} = \eta^{(2)}_{3}+\eta^{(2)}_{6}}

（第 $2$ 层，$\ell = 2$。）

\fitmath{\eta^{(2)}_{1} = \eta^{(3)}_{1}+\eta^{(3)}_{4}}

\fitmath{\eta^{(2)}_{2} = \eta^{(3)}_{2}+\eta^{(3)}_{8}}

\fitmath{\eta^{(2)}_{3} = \eta^{(3)}_{3}+\eta^{(3)}_{5}}

\fitmath{\eta^{(2)}_{6} = \eta^{(3)}_{6}+\eta^{(3)}_{7}}

（第 $3$ 层，$\ell = 2$。）

\section*{辅助单位根}
$\ell = 2$ 时 $\omega_{2}^{1} = -1$。

$\ell \geq 3$ 时按主支：

\fitmath{\omega_{\ell}^{k} = \frac{\Lambda^{(\ell)}_{\bar k}}{2} + \frac{\varepsilon_k}{2}\sqrt{2-\Lambda^{(\ell)}_{\overline{2k}}}\,i}

其中 $\varepsilon_k = \operatorname{sgn}\sin(2\pi k/\ell)$，$\bar k = \min(k,\ell-k)$。

\section*{逐层开方}
每一步把母周期 $\eta^{(j-1)}_{a}$ 分解成 $\ell$ 个子周期 $\beta^{(a)}_{t}$。

预解式取离散 Fourier 变换：

\fitmath{\theta^{(j)}_{a,k} = \sum_{t=0}^{\ell-1}\omega_{\ell}^{-kt}\beta^{(a)}_{t}}

\fitmath{\left(\theta^{(j)}_{a,k}\right)^{\ell} = u^{(j)}_{a,k} = \tau^{k}\left(u^{(j)}_{a,1}\right)}

\fitmath{\eta^{(j)}_{b(t)} = \frac{1}{\ell}\left(\eta^{(j-1)}_{a} + \sum_{k=1}^{\ell-1}\omega_{\ell}^{kt}\theta^{(j)}_{a,k}\right)}

（$\tau$ 把 $\omega_{\ell}$ 送到 $\omega_{\ell}^{k}$；$b(t)$ 为第 $t$ 个子周期。）

开方一律取正实主支；预解式的分支由因子 $\omega_{\ell}^{s}$ 记录（见报告里的分支表）。

\subsection*{第 $1$ 步（$\ell = 2$，1 个母周期）}
\fitmath{u^{(1)}_{1,1} = 17}

\fitmath{\theta^{(1)}_{1,1} = \sqrt{17}}

\fitmath{\theta^{(2)}_{3,1} = \frac{1}{2}\sqrt{34 + 2\sqrt{17}}}

\fitmath{\theta^{(3)}_{1,1} = \sqrt{\frac{17}{4} + \frac{3}{4}\sqrt{17} - \frac{1}{4}\sqrt{34 - 2\sqrt{17}} - \frac{1}{2}\sqrt{34 + 2\sqrt{17}}}}

\fitmath{\theta^{(3)}_{2,1} = \sqrt{\frac{17}{4} + \frac{3}{4}\sqrt{17} + \frac{1}{4}\sqrt{34 - 2\sqrt{17}} + \frac{1}{2}\sqrt{34 + 2\sqrt{17}}}}

\fitmath{u^{(3)}_{3,1} = \frac{17}{4}-\frac{3}{4}\sqrt{17}+\frac{1}{2}\sqrt{34 - 2\sqrt{17}}-\frac{1}{4}\sqrt{34 + 2\sqrt{17}}}

\fitmath{\theta^{(3)}_{3,1} = \sqrt{\frac{17}{4} - \frac{3}{4}\sqrt{17} + \frac{1}{2}\sqrt{34 - 2\sqrt{17}} - \frac{1}{4}\sqrt{34 + 2\sqrt{17}}}}

\fitmath{\theta^{(3)}_{6,1} = \sqrt{\frac{17}{4} - \frac{3}{4}\sqrt{17} - \frac{1}{2}\sqrt{34 - 2\sqrt{17}} + \frac{1}{4}\sqrt{34 + 2\sqrt{17}}}}
