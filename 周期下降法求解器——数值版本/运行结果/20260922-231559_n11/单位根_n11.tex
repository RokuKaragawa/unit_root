\section*{参数与标准素分解}
$p = 11$，$h = (p-1)/2 = 5$，模 $11$ 的最小原根 $g = 2$。

$h$ 的标准素分解：$h = 5$，子群链共 $1$ 层。

$m_0 = 1$，$m_{1} = 5$。

各层子群 $H_j = \{\pm g^{m_j t}\}$，其中 $H_0$ 是全部 $10$ 个非零剩余：

\fitmath{H_{1} = \{1,\,10\}}

各层周期 $\eta^{(j)}_{a} = \sum_{b \in H_j}\zeta^{ab}$（$a$ 取陪集代表）。

\section*{各层周期与分解表}
\subsection*{第 $0$ 层：1 个周期}
$H_0$ 是全部 $10$ 个非零剩余，故 $\eta^{(0)}_{1} = -1$。

\subsection*{第 $1$ 层：5 个周期}
\fitmath{\eta^{(1)}_{1} = \zeta^{1}+\zeta^{10}}

\fitmath{\eta^{(1)}_{2} = \zeta^{2}+\zeta^{9}}

\fitmath{\eta^{(1)}_{3} = \zeta^{3}+\zeta^{8}}

\fitmath{\eta^{(1)}_{4} = \zeta^{4}+\zeta^{7}}

\fitmath{\eta^{(1)}_{5} = \zeta^{5}+\zeta^{6}}

\subsection*{分解表：每个母周期拆成 $\ell_j$ 个子周期}
\fitmath{\eta^{(0)}_{1} = \eta^{(1)}_{1}+\eta^{(1)}_{2}+\eta^{(1)}_{4}+\eta^{(1)}_{3}+\eta^{(1)}_{5}}

（第 $1$ 层，$\ell = 5$。）

\section*{辅助单位根}
$\ell = 2$ 时 $\omega_{2}^{1} = -1$。

$\ell \geq 3$ 时按主支：

\fitmath{\omega_{\ell}^{k} = \frac{\Lambda^{(\ell)}_{\bar k}}{2} + \frac{\varepsilon_k}{2}\sqrt{2-\Lambda^{(\ell)}_{\overline{2k}}}\,i}

其中 $\varepsilon_k = \operatorname{sgn}\sin(2\pi k/\ell)$，$\bar k = \min(k,\ell-k)$。

\subsection*{辅助问题 $p = 5$（供 $\omega_{5}^{k}$ 使用）}
$h = 2$，$g = 2$，层阶 $2$。

\fitmath{\Lambda^{(5)}_{1} = \frac{-1 + \sqrt{5}}{2}}

\fitmath{\Lambda^{(5)}_{2} = \frac{-1 - \sqrt{5}}{2}}

$\omega_{5}^{k} = \frac{1}{2}\Lambda^{(5)}_{\bar k} + \frac{\varepsilon_k}{2}\sqrt{2-\Lambda^{(5)}_{\overline{2k}}}\,i$。

\section*{逐层开方}
每一步把母周期 $\eta^{(j-1)}_{a}$ 分解成 $\ell$ 个子周期 $\beta^{(a)}_{t}$。

预解式取离散 Fourier 变换：

\fitmath{\theta^{(j)}_{a,k} = \sum_{t=0}^{\ell-1}\omega_{\ell}^{-kt}\beta^{(a)}_{t}}

\fitmath{\left(\theta^{(j)}_{a,k}\right)^{\ell} = u^{(j)}_{a,k} = \tau^{k}\left(u^{(j)}_{a,1}\right)}

\fitmath{\eta^{(j)}_{b(t)} = \frac{1}{\ell}\left(\eta^{(j-1)}_{a} + \sum_{k=1}^{\ell-1}\omega_{\ell}^{kt}\theta^{(j)}_{a,k}\right)}

（$\tau$ 把 $\omega_{\ell}$ 送到 $\omega_{\ell}^{k}$；$b(t)$ 为第 $t$ 个子周期。）

开方一律取正实主支；预解式的分支由因子 $\omega_{\ell}^{s}$ 记录（见报告里的分支表）。
