\section*{参数与标准素分解}
$p = 13$，$h = (p-1)/2 = 6$，模 $13$ 的最小原根 $g = 2$。

$h$ 的标准素分解：$h = 3\times2$，子群链共 $2$ 层。

$m_0 = 1$，$m_{1} = 3$，$m_{2} = 6$。

各层子群 $H_j = \{\pm g^{m_j t}\}$，其中 $H_0$ 是全部 $12$ 个非零剩余：

\fitmath{H_{1} = \{1,\,5,\,8,\,12\}}

\fitmath{H_{2} = \{1,\,12\}}

各层周期 $\eta^{(j)}_{a} = \sum_{b \in H_j}\zeta^{ab}$（$a$ 取陪集代表）。

\section*{各层周期与分解表}
\subsection*{第 $0$ 层：1 个周期}
$H_0$ 是全部 $12$ 个非零剩余，故 $\eta^{(0)}_{1} = -1$。

\subsection*{第 $1$ 层：3 个周期}
\fitmath{\eta^{(1)}_{1} = \zeta^{1}+\zeta^{5}+\zeta^{8}+\zeta^{12}}

\fitmath{\eta^{(1)}_{2} = \zeta^{2}+\zeta^{3}+\zeta^{10}+\zeta^{11}}

\fitmath{\eta^{(1)}_{4} = \zeta^{4}+\zeta^{6}+\zeta^{7}+\zeta^{9}}

\subsection*{第 $2$ 层：6 个周期}
\fitmath{\eta^{(2)}_{1} = \zeta^{1}+\zeta^{12}}

\fitmath{\eta^{(2)}_{2} = \zeta^{2}+\zeta^{11}}

\fitmath{\eta^{(2)}_{3} = \zeta^{3}+\zeta^{10}}

\fitmath{\eta^{(2)}_{4} = \zeta^{4}+\zeta^{9}}

\fitmath{\eta^{(2)}_{5} = \zeta^{5}+\zeta^{8}}

\fitmath{\eta^{(2)}_{6} = \zeta^{6}+\zeta^{7}}

\subsection*{分解表：每个母周期拆成 $\ell_j$ 个子周期}
\fitmath{\eta^{(0)}_{1} = \eta^{(1)}_{1}+\eta^{(1)}_{2}+\eta^{(1)}_{4}}

（第 $1$ 层，$\ell = 3$。）

\fitmath{\eta^{(1)}_{1} = \eta^{(2)}_{1}+\eta^{(2)}_{5}}

\fitmath{\eta^{(1)}_{2} = \eta^{(2)}_{2}+\eta^{(2)}_{3}}

\fitmath{\eta^{(1)}_{4} = \eta^{(2)}_{4}+\eta^{(2)}_{6}}

（第 $2$ 层，$\ell = 2$。）

\section*{辅助单位根}
$\ell = 2$ 时 $\omega_{2}^{1} = -1$。

$\ell \geq 3$ 时按主支：

\fitmath{\omega_{\ell}^{k} = \frac{\Lambda^{(\ell)}_{\bar k}}{2} + \frac{\varepsilon_k}{2}\sqrt{2-\Lambda^{(\ell)}_{\overline{2k}}}\,i}

其中 $\varepsilon_k = \operatorname{sgn}\sin(2\pi k/\ell)$，$\bar k = \min(k,\ell-k)$。

\subsection*{辅助问题 $p = 3$（供 $\omega_{3}^{k}$ 使用）}
$h = 1$，$g = 2$，层阶 $1$。

\fitmath{\Lambda^{(3)}_{1} = -1}

$\omega_{3}^{k} = \frac{1}{2}\Lambda^{(3)}_{\bar k} + \frac{\varepsilon_k}{2}\sqrt{2-\Lambda^{(3)}_{\overline{2k}}}\,i$。

\section*{逐层开方}
每一步把母周期 $\eta^{(j-1)}_{a}$ 分解成 $\ell$ 个子周期 $\beta^{(a)}_{t}$。

预解式取离散 Fourier 变换：

\fitmath{\theta^{(j)}_{a,k} = \sum_{t=0}^{\ell-1}\omega_{\ell}^{-kt}\beta^{(a)}_{t}}

\fitmath{\left(\theta^{(j)}_{a,k}\right)^{\ell} = u^{(j)}_{a,k} = \tau^{k}\left(u^{(j)}_{a,1}\right)}

\fitmath{\eta^{(j)}_{b(t)} = \frac{1}{\ell}\left(\eta^{(j-1)}_{a} + \sum_{k=1}^{\ell-1}\omega_{\ell}^{kt}\theta^{(j)}_{a,k}\right)}

（$\tau$ 把 $\omega_{\ell}$ 送到 $\omega_{\ell}^{k}$；$b(t)$ 为第 $t$ 个子周期。）

开方一律取正实主支；预解式的分支由因子 $\omega_{\ell}^{s}$ 记录（见报告里的分支表）。
